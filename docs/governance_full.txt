Section 3: Guardian Monitor Handler
Create server/scheduledHandlers/guardianMonitor.ts:
/** * Guardian Monitor — Runs every 5 minutes * Checks all 7 guardrails and records status. */import type { Request, Response } from "express";import { sdk } from "../_core/sdk";import { getDb } from "../db";import { sql } from "drizzle-orm";export async function guardianMonitorHandler(req: Request, res: Response) {  // Authenticate cron request  try {    const user = await sdk.authenticateRequest(req);    if (!user.isCron) {      const ip = req.ip ?? "";      const isInternal = ip === "127.0.0.1" || ip === "::1" || ip.startsWith("::ffff:127.");      if (!isInternal) return res.status(403).json({ error: "cron-only endpoint" });    }  } catch {    const ip = req.ip ?? "";    const isInternal = ip === "127.0.0.1" || ip === "::1" || ip.startsWith("::ffff:127.");    if (!isInternal) return res.status(401).json({ error: "Unauthorized" });  }  const db = await getDb();  if (!db) return res.status(500).json({ error: "Database unavailable" });  const results: Record<string, { status: string; issues: number; details?: any }> = {};  try {    // GR-3: Shadow block backlog check    const backlogResult = await db.execute(sql`      SELECT         COUNT(*) as total_pending,        SUM(CASE WHEN syncStatus = 'sync_failed' THEN 1 ELSE 0 END) as failed_count,        SUM(CASE WHEN createdAt < NOW() - INTERVAL 24 HOUR THEN 1 ELSE 0 END) as stale_count      FROM shadow_blocks       WHERE syncStatus IN ('pending_sync', 'sync_failed')    `);    const backlog = (Array.isArray(backlogResult) ? backlogResult[0] : backlogResult)?.[0];    results["GR-3"] = {      status: backlog?.stale_count > 0 ? "warning" : "healthy",      issues: Number(backlog?.stale_count || 0),      details: { totalPending: Number(backlog?.total_pending || 0), failedCount: Number(backlog?.failed_count || 0) },    };    // GR-4: Calendar sync health — check for expired tokens    const tokenResult = await db.execute(sql`      SELECT         COUNT(*) as total_tokens,        SUM(CASE WHEN status != 'active' THEN 1 ELSE 0 END) as expired_tokens,        MAX(lastSyncedAt) as last_sync      FROM oauth_tokens       WHERE provider = 'google'    `);    const tokenHealth = (Array.isArray(tokenResult) ? tokenResult[0] : tokenResult)?.[0];    const hoursSinceSync = tokenHealth?.last_sync       ? (Date.now() - new Date(tokenHealth.last_sync).getTime()) / 3600000       : 999;    results["GR-4"] = {      status: tokenHealth?.expired_tokens > 0 ? "warning" : hoursSinceSync > 24 ? "critical" : "healthy",      issues: Number(tokenHealth?.expired_tokens || 0),      details: { hoursSinceSync: Math.round(hoursSinceSync), totalTokens: Number(tokenHealth?.total_tokens || 0) },    };    // GR-5: Financial anomaly — check for bookings with unusual amounts    const anomalyResult = await db.execute(sql`      SELECT COUNT(*) as anomaly_count      FROM bookings b      LEFT JOIN platform_fee_configurations pfc         ON b.householdId = pfc.householdId AND b.platform = pfc.platform      WHERE b.hostPayout IS NOT NULL         AND b.totalAmount IS NOT NULL        AND b.hostPayout > b.totalAmount * 0.95  -- Payout > 95% of total is suspicious        AND b.createdAt > NOW() - INTERVAL 7 DAY    `);    const anomaly = (Array.isArray(anomalyResult) ? anomalyResult[0] : anomalyResult)?.[0];    results["GR-5"] = {      status: anomaly?.anomaly_count > 0 ? "warning" : "healthy",      issues: Number(anomaly?.anomaly_count || 0),    };    // Record the health check    await db.execute(sql`      INSERT INTO guardian_audit_log (id, guardrailId, severity, action, details, createdAt)      VALUES (        UUID(),        'SYSTEM',        ${results["GR-4"].status === "critical" || results["GR-3"].issues > 100 ? "critical" : "info"},        'monitor_check',        ${JSON.stringify(results)},        NOW()      )    `);    // If critical issues found, trigger escalation    const hasCritical = Object.values(results).some(r => r.status === "critical");    if (hasCritical) {      await escalateCriticalAlert(results, db);    }    return res.json({ status: hasCritical ? "critical" : "healthy", guardrails: results, checkedAt: new Date().toISOString() });  } catch (err) {    console.error("[GuardianMonitor] Error:", err);    return res.status(500).json({ error: (err as Error)?.message });  }}async function escalateCriticalAlert(results: any, db: any) {  // Send email alert via Resend  const resendKey = process.env.RESEND_API_KEY;  if (!resendKey) return;  try {    await fetch("https://api.resend.com/emails", {      method: "POST",      headers: { Authorization: `Bearer ${resendKey}`, "Content-Type": "application/json" },      body: JSON.stringify({        from: "Geeves.Life Guardian <alerts@geeves.life>",        to: [process.env.GUARDIAN_ALERT_EMAIL || "tarik@tjperkinsfam.com"],        subject: "🚨 Geeves Guardian: Critical Issues Detected",        html: `<h2>Guardian Critical Alert</h2><pre>${JSON.stringify(results, null, 2)}</pre><p>Dashboard: <a href="https://beta.geeves.life/super-admin?tab=guardian">View Details</a></p>`,      }),    });  } catch (e) {    console.error("[GuardianMonitor] Failed to send alert email:", e);  }}

Section 4: Guardian Daily Digest Handler
Create server/scheduledHandlers/guardianDailyDigest.ts:
/** * Guardian Daily Digest — Runs every day at 08:00 * Sends a summary email of all guardrail activity in the last 24 hours. */import type { Request, Response } from "express";import { sdk } from "../_core/sdk";import { getDb } from "../db";import { sql } from "drizzle-orm";export async function guardianDailyDigestHandler(req: Request, res: Response) {  try {    const user = await sdk.authenticateRequest(req);    if (!user.isCron) {      const ip = req.ip ?? "";      const isInternal = ip === "127.0.0.1" || ip === "::1" || ip.startsWith("::ffff:127.");      if (!isInternal) return res.status(403).json({ error: "cron-only endpoint" });    }  } catch {    const ip = req.ip ?? "";    const isInternal = ip === "127.0.0.1" || ip === "::1" || ip.startsWith("::ffff:127.");    if (!isInternal) return res.status(401).json({ error: "Unauthorized" });  }  const db = await getDb();  if (!db) return res.status(500).json({ error: "Database unavailable" });  try {    // Get 24h summary    const summaryResult = await db.execute(sql`      SELECT         guardrailId,        action,        severity,        COUNT(*) as count      FROM guardian_audit_log      WHERE createdAt > NOW() - INTERVAL 24 HOUR      GROUP BY guardrailId, action, severity      ORDER BY FIELD(severity, 'emergency', 'critical', 'warning', 'info'), count DESC    `);    const summary = Array.isArray(summaryResult) ? summaryResult[0] : summaryResult;    // Get unresolved critical issues    const criticalResult = await db.execute(sql`      SELECT id, guardrailId, details, createdAt      FROM guardian_audit_log      WHERE severity IN ('critical', 'emergency')        AND resolvedAt IS NULL        AND createdAt > NOW() - INTERVAL 7 DAY      ORDER BY createdAt DESC      LIMIT 20    `);    const critical = Array.isArray(criticalResult) ? criticalResult[0] : criticalResult;    // Send digest email    const resendKey = process.env.RESEND_API_KEY;    if (resendKey) {      await fetch("https://api.resend.com/emails", {        method: "POST",        headers: { Authorization: `Bearer ${resendKey}`, "Content-Type": "application/json" },        body: JSON.stringify({          from: "Geeves.Life Guardian <reports@geeves.life>",          to: [process.env.GUARDIAN_ALERT_EMAIL || "tarik@tjperkinsfam.com"],          subject: `Geeves Guardian Daily Digest — ${new Date().toLocaleDateString()}`,          html: `<h2>Geeves Guardian — Daily Digest</h2>            <h3>Last 24 Hours Summary</h3>            <pre>${JSON.stringify(summary, null, 2)}</pre>            <h3>Unresolved Critical Issues (${(critical as any[])?.length || 0})</h3>            <pre>${JSON.stringify(critical, null, 2)}</pre>            <p><a href="https://beta.geeves.life/super-admin?tab=guardian">View Guardian Dashboard</a></p>`,        }),      });    }    return res.json({ summary, criticalCount: (critical as any[])?.length || 0, sentAt: new Date().toISOString() });  } catch (err) {    console.error("[GuardianDigest] Error:", err);    return res.status(500).json({ error: (err as Error)?.message });  }}

Section 5: Guardian Financial Sweep Handler
Create server/scheduledHandlers/guardianFinancialSweep.ts:
/** * Guardian Financial Sweep — Runs every 30 minutes * Auto-reconciles recent bookings and flags anomalies. */import type { Request, Response } from "express";import { sdk } from "../_core/sdk";import { getDb } from "../db";import { sql } from "drizzle-orm";export async function guardianFinancialSweepHandler(req: Request, res: Response) {  try {    const user = await sdk.authenticateRequest(req);    if (!user.isCron) {      const ip = req.ip ?? "";      const isInternal = ip === "127.0.0.1" || ip === "::1" || ip.startsWith("::ffff:127.");      if (!isInternal) return res.status(403).json({ error: "cron-only endpoint" });    }  } catch {    const ip = req.ip ?? "";    const isInternal = ip === "127.0.0.1" || ip === "::1" || ip.startsWith("::ffff:127.");    if (!isInternal) return res.status(401).json({ error: "Unauthorized" });  }  const db = await getDb();  if (!db) return res.status(500).json({ error: "Database unavailable" });  try {    // Find bookings in the last 48 hours that haven't been reconciled    const unreconciledResult = await db.execute(sql`      SELECT b.id, b.householdId, b.platform, b.totalAmount, b.hostPayout, b.checkIn      FROM bookings b      LEFT JOIN guardian_audit_log gal         ON gal.targetId = b.id AND gal.guardrailId = 'GR-5' AND gal.createdAt > b.updatedAt      WHERE b.hostPayout IS NOT NULL         AND b.totalAmount IS NOT NULL        AND b.createdAt > NOW() - INTERVAL 48 HOUR        AND gal.id IS NULL      LIMIT 100    `);    const bookings = Array.isArray(unreconciledResult) ? unreconciledResult[0] : unreconciledResult;    let flagged = 0;    let cleared = 0;    for (const booking of (bookings as any[]) || []) {      // Get platform fee config      const configResult = await db.execute(sql`        SELECT commissionPct, processingFeePct, processingFeeFixed, reconciliationTolerancePct        FROM platform_fee_configurations        WHERE householdId = ${booking.householdId} AND platform = ${booking.platform} AND isActive = TRUE        LIMIT 1      `);      const config = (Array.isArray(configResult) ? configResult[0] : configResult)?.[0];      const gross = Number(booking.totalAmount);      const expectedFee = config        ? gross * (Number(config.commissionPct) / 100) + gross * (Number(config.processingFeePct) / 100) + Number(config.processingFeeFixed)        : gross * 0.155; // Default Airbnb      const expectedNet = gross - expectedFee;      const actualNet = Number(booking.hostPayout);      const discrepancyPct = actualNet > 0 ? Math.abs((expectedNet - actualNet) / actualNet) * 100 : 0;      const tolerance = Number(config?.reconciliationTolerancePct || 2);      if (discrepancyPct > tolerance * 2) {        // Flag for review        await db.execute(sql`          INSERT INTO guardian_audit_log (id, guardrailId, severity, action, targetId, details, createdAt)          VALUES (UUID(), 'GR-5', 'warning', 'flagged', ${booking.id},             ${JSON.stringify({ expectedNet, actualNet, discrepancyPct, platform: booking.platform })}, NOW())        `);        flagged++;      } else {        cleared++;      }    }    return res.json({ processed: (bookings as any[])?.length || 0, flagged, cleared, checkedAt: new Date().toISOString() });  } catch (err) {    console.error("[GuardianFinancial] Error:", err);    return res.status(500).json({ error: (err as Error)?.message });  }}

Section 8: Emergency Procedures
If Guardian triggers emergency shutdown (GR-7):
Check guardian_audit_log for the trigger reason
Review the Super Admin dashboard at /super-admin?tab=guardian
Resolve the underlying issue
Run: UPDATE guardian_config SET gr7AutoFix = FALSE WHERE householdId = '<id>';
If manual override needed: UPDATE guardian_audit_log SET resolvedAt = NOW(), resolvedBy = '<your-user-id>' WHERE id = '<alert-id>';
If cron jobs stop running:
Check Manus dashboard for cron job status
Verify SYSTEM_CRON_SECRET matches in both env and dashboard
Check server logs: grep "cron-only endpoint" /var/log/geeves.log
Test endpoint manually with curl (see Section 7)
If alert emails not received:
Verify RESEND_API_KEY is set and valid
Check GUARDIAN_ALERT_EMAIL is set correctly
Check Resend dashboard for delivery status
Check spam/junk folders
